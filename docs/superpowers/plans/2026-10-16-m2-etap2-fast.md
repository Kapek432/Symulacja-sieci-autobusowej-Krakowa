```python
import numpy as np
import pandas as pd
import pytest

from krakbus.observed.events import index_schedule
from krakbus.sim.boundary import boundary_starts
from krakbus.sim.engine import run_simpy
from krakbus.sim.fast import run_fast
from krakbus.sim.network import Network, build_network, select_trips
from krakbus.sim.noise import draw_noise
from krakbus.sim.params import EtaParams, LoopParams, ModelParams, UParams

from .conftest import DATE, TZ, make_schedule, row, zero_model

NOISY = zero_model(
    drift_s_per_min={"szczyt_rano": 4.0, "poza": 3.0},
    day_effect_sd_s=5.0,
    u=UParams(spatial="independent", window_s=900, sigma_s=10.0, rho=0.7),
    eta=EtaParams(df=3.0, scale_s=40.0),
    loop=LoopParams(l_min_s=60.0, delta_mean_s=20.0, delta_sd_s=20.0),
)


def _hhmm(minute: int) -> str:
    return f"{minute // 60:02d}:{minute % 60:02d}"


def _loop_network() -> Network:
    """40 blokow po 6 kursow, zapas 8 min, co trzeci kurs na linii B (przejscia miedzy liniami)."""
    stops = [f"p{k}" for k in range(5)]
    trips = []
    for b in range(40):
        for k in range(6):
            seq = stops if k % 2 == 0 else stops[::-1]
            t0 = 300 + 2 * b + 20 * k
            times = [(s, _hhmm(t0 + 3 * j)) for j, s in enumerate(seq)]
            route = "B" if k % 3 == 2 else "A"
            trips.append((f"b{b}k{k}", route, str(k % 2), f"blk{b}", times))
    return build_network(index_schedule(make_schedule(trips)), DATE, TZ)


def _both(net: Network, model: ModelParams, seed: int, routes: list[str] | None, obs: pd.Series | None):
    sel = select_trips(net, routes)
    noise = draw_noise(net, model, seed)
    starts = boundary_starts(net, sel, noise, model, obs)
    return run_simpy(net, sel, noise, model, starts), run_fast(net, sel, noise, model, starts)


@pytest.mark.parametrize("routes", [["A"], ["B"], None])
@pytest.mark.parametrize("seed", [0, 1, 2])
def test_fast_equals_simpy_on_small_net(small_net, routes, seed) -> None:
    obs = pd.Series({"B1": 400.0, "A0": 30.0})
    a, b = _both(small_net, NOISY, seed, routes, obs)
    np.testing.assert_allclose(b, a, atol=1e-6, equal_nan=True)


@pytest.mark.parametrize("routes", [["A"], None])
def test_fast_equals_simpy_with_loops_and_cross_route(routes) -> None:
    net = _loop_network()
    obs = pd.Series(250.0, index=net.trips.index)
    a, b = _both(net, NOISY, 7, routes, obs)
    np.testing.assert_allclose(b, a, atol=1e-6, equal_nan=True)
    assert np.isnan(a).any() == (routes is not None)


def test_fast_carries_and_absorbs_like_loop_rule(small_net) -> None:
    m = zero_model()
    sel = select_trips(small_net, None)
    for shock, expected in ((100.0, 0.0), (500.0, 260.0)):
        noise = draw_noise(small_net, m, 0)
        noise.eta[row(small_net, "A0", "s3")] = shock
        d = run_fast(small_net, sel, noise, m, {})
        assert d[row(small_net, "A0", "s4")] == shock
        assert d[row(small_net, "A1", "s1")] == expected


def test_fast_refuses_gamma(small_net) -> None:
    m = zero_model(gamma=0.5)
    sel = select_trips(small_net, None)
    with pytest.raises(ValueError, match="gamma"):
        run_fast(small_net, sel, draw_noise(small_net, m, 0), m, {})
```

```python
"""Szybka sciezka doby (numpy) dla gamma = 0: rekursja po pozycji kursu w lancuchu."""

import numpy as np

from krakbus.sim.model import base_increments, loop_departure
from krakbus.sim.network import Network, chain_order
from krakbus.sim.noise import Noise
from krakbus.sim.params import ModelParams


def _offsets(inc: np.ndarray, trip_of_row: np.ndarray, row_start: np.ndarray) -> np.ndarray:
    """Suma przyrostow od poczatku kursu do wiersza, bez przyrostu z samego wiersza."""
    before = np.cumsum(inc) - inc
    return before - before[row_start[trip_of_row]]


def run_fast(
    net: Network,
    sel: np.ndarray,
    noise: Noise,
    model: ModelParams,
    starts: dict[int, float],
) -> np.ndarray:
    """Doba dla kursow z wyboru bez SimPy; wynik jak `run_simpy` przy gamma = 0.

    Przy gamma = 0 kurs zalezy od innych tylko przez poprzednika w lancuchu, wiec opoznienie
    w wierszu to opoznienie odjazdu kursu plus suma przyrostow od jego poczatku. Petla idzie
    po pozycji w lancuchu: na pozycji k wszystkie kursy dostaja odjazd z reguly petli
    zastosowanej do kursow z pozycji k - 1.

    Args:
        net: siec dnia.
        sel: pozycje kursow w zakresie.
        noise: szum przebiegu.
        model: parametry modelu.
        starts: Δ0 kursow z poprzednikiem spoza zakresu (`boundary_starts`).

    Returns:
        Opoznienie przyjazdu (= odjazdu) dla kazdego wiersza `net.stop_times`; NaN poza wyborem.

    Raises:
        ValueError: gamma != 0.
    """
    if model.gamma != 0:
        raise ValueError("szybka sciezka liczy tylko gamma = 0; dla gamma != 0 silnik simpy")
    trips = net.trips
    row_start = trips["row_start"].to_numpy()
    n_rows = trips["n_rows"].to_numpy()
    trip_of_row = np.repeat(np.arange(len(trips)), n_rows)
    off = _offsets(base_increments(net, noise, model), trip_of_row, row_start)
    last_row = row_start + n_rows - 1
    layover = trips["layover_in_s"].to_numpy(float)
    start = noise.delta.astype(float).copy()
    for p, d0 in starts.items():
        start[p] = d0
    order = chain_order(net, sel)
    trip = order["trip"].to_numpy()
    at_pos = order["pos"].to_numpy()
    prev = order["prev_sel"].to_numpy()
    dep = np.full(len(trips), np.nan)
    for k in range(int(at_pos.max()) + 1 if len(at_pos) else 0):
        here = at_pos == k
        p, q = trip[here], prev[here]
        d = start[p]
        inner = q >= 0
        end = dep[q[inner]] + off[last_row[q[inner]]]
        d[inner] = loop_departure(end, layover[p[inner]], model.loop.l_min_s, noise.delta[p[inner]])
        dep[p] = d
    delay = np.full(len(net.stop_times), np.nan)
    simulated = ~np.isnan(dep)[trip_of_row]
    delay[simulated] = dep[trip_of_row[simulated]] + off[simulated]
    return delay
```
