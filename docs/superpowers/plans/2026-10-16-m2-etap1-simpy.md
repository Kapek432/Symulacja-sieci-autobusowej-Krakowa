```python
"""Parametry symulacji z `configs/sim.yaml`."""

from dataclasses import dataclass
from pathlib import Path

import yaml

from krakbus.config import ROOT

DEFAULT_SIM_CONFIG = ROOT / "configs" / "sim.yaml"
OFF_PEAK = "poza"
SPATIAL = ("independent",)
BOUNDARY = ("observed", "delta")


@dataclass(frozen=True)
class UParams:
    spatial: str
    window_s: int
    sigma_s: float
    rho: float


@dataclass(frozen=True)
class EtaParams:
    df: float
    scale_s: float


@dataclass(frozen=True)
class LoopParams:
    l_min_s: float
    delta_mean_s: float
    delta_sd_s: float


@dataclass(frozen=True)
class ModelParams:
    periods: dict[str, tuple[int, int]]
    drift_s_per_min: dict[str, float]
    day_effect_sd_s: float
    u: UParams
    eta: EtaParams
    loop: LoopParams
    gamma: float


@dataclass(frozen=True)
class SimConfig:
    scopes: dict[str, list[str] | None]
    model: ModelParams
    boundary: str
    seed: int


def _model(raw: dict, periods: dict[str, tuple[int, int]]) -> ModelParams:
    u, eta, loop = raw["u"], raw["eta"], raw["loop"]
    return ModelParams(
        periods=periods,
        drift_s_per_min={k: float(v) for k, v in raw["drift_s_per_min"].items()},
        day_effect_sd_s=float(raw["day_effect_sd_s"]),
        u=UParams(
            spatial=str(u["spatial"]),
            window_s=int(u["window_s"]),
            sigma_s=float(u["sigma_s"]),
            rho=float(u["rho"]),
        ),
        eta=EtaParams(df=float(eta["df"]), scale_s=float(eta["scale_s"])),
        loop=LoopParams(
            l_min_s=float(loop["l_min_s"]),
            delta_mean_s=float(loop["delta_mean_s"]),
            delta_sd_s=float(loop["delta_sd_s"]),
        ),
        gamma=float(raw["gamma"]),
    )


def load_sim_config(path: Path = DEFAULT_SIM_CONFIG) -> SimConfig:
    """Wczytuje i sprawdza config symulacji.

    Args:
        path: plik YAML z sekcjami `scopes`, `periods`, `model`, `boundary`, `seed`.

    Returns:
        Niezmienny obiekt configu.

    Raises:
        ValueError: nieznany wariant `u.spatial` albo `boundary`, brak dryfu dla pory dnia.
    """
    raw = yaml.safe_load(path.read_text(encoding="utf-8"))
    periods = {k: (int(v[0]), int(v[1])) for k, v in raw["periods"].items()}
    model = _model(raw["model"], periods)
    if model.u.spatial not in SPATIAL:
        raise ValueError(f"u.spatial = {model.u.spatial}; dostepne: {SPATIAL}")
    if raw["boundary"] not in BOUNDARY:
        raise ValueError(f"boundary = {raw['boundary']}; dostepne: {BOUNDARY}")
    missing = (set(periods) | {OFF_PEAK}) - set(model.drift_s_per_min)
    if missing:
        raise ValueError(f"brak drift_s_per_min dla por: {sorted(missing)}")
    scopes = {
        k: None if v is None else [str(r) for r in v] for k, v in raw["scopes"].items()
    }
    return SimConfig(scopes=scopes, model=model, boundary=raw["boundary"], seed=int(raw["seed"]))
```

```python
def make_schedule(trips: list[tuple[str, str, str, str, list[tuple[str, str]]]]) -> Schedule:
    tr = pd.DataFrame(
        [(t, r, d, b) for t, r, d, b, _ in trips],
        columns=["trip_id", "route_id", "direction_id", "block_id"],
    ).assign(service_id="svc", gtfs_stamp="20261012_000000")
    rows = [
        (t, str(seq), stop, f"{hhmm}:00", f"{hhmm}:00")
        for t, _, _, _, stops in trips
        for seq, (stop, hhmm) in enumerate(stops, start=1)
    ]
```

```python
"""Siec doby obslugi: kursy wszystkich linii, przystanki, odcinki i lancuchy (bloki)."""

from dataclasses import dataclass

import numpy as np
import pandas as pd

from krakbus.ingest.gtfs_versions import gtfs_day_start
from krakbus.observed.events import ScheduleIndex


@dataclass(frozen=True)
class Network:
    """Siec jednego dnia obslugi w postaci tablic.

    Attributes:
        service_date: data obslugi YYYY-MM-DD.
        day_start: poczatek doby GTFS (lokalne poludnie minus 12 h) w sekundach epoki.
        trips: indeks `trip_id` w kolejnosci (`block_key`, `start_s`); `route_id`, `direction_id`,
            `block_key`, `blk_idx`, `start_s`, `end_s`, `prev_trip` (poprzedni kurs bloku),
            `layover_in_s` (start minus koniec poprzednika), `row_start`, `n_rows`.
        stop_times: wiersze kursow w kolejnosci `trips` i `stop_sequence`; `trip_id`,
            `stop_sequence`, `stop_id`, `arr_s`, `dep_s`, `t_s` (rozkladowy czas odcinka
            wychodzacego, 0 na ostatnim przystanku), `seg_id` (-1 na ostatnim przystanku).
        segments: indeks `seg_id`; `from_stop`, `to_stop` (skierowany odcinek).
    """

    service_date: str
    day_start: float
    trips: pd.DataFrame
    stop_times: pd.DataFrame
    segments: pd.DataFrame


def _day_trips(idx: ScheduleIndex, ymd: str) -> pd.DataFrame:
    services = set(idx.calendar.loc[idx.calendar["date"] == ymd, "service_id"])
    plan = idx.trip_plan[idx.trip_plan["service_id"].isin(services)].reset_index()
    if plan.empty:
        raise ValueError(f"brak aktywnych kursow w dniu {ymd}")
    block = plan["block_id"].astype("string").fillna(plan["trip_id"])
    plan["block_key"] = plan["service_id"].astype(str) + "|" + block
    plan["start_s"] = plan["start_s"].astype(float)
    plan["end_s"] = plan["end_s"].astype(float)
    t = plan.sort_values(["block_key", "start_s", "trip_id"]).reset_index(drop=True)
    g = t.groupby("block_key", sort=False)
    t["blk_idx"] = g.cumcount()
    t["prev_trip"] = g["trip_id"].shift(1)
    t["layover_in_s"] = t["start_s"] - g["end_s"].shift(1)
    cols = ["trip_id", "route_id", "direction_id", "block_key", "blk_idx", "start_s", "end_s"]
    return t[cols + ["prev_trip", "layover_in_s"]]


def _day_stop_times(idx: ScheduleIndex, trips: pd.DataFrame) -> pd.DataFrame:
    order = pd.Series(np.arange(len(trips)), index=trips["trip_id"])
    st = idx.stop_times[idx.stop_times["trip_id"].isin(order.index)]
    st = st.assign(_o=st["trip_id"].map(order)).sort_values(["_o", "stop_sequence"])
    st = st.drop(columns="_o").reset_index(drop=True)
    st["arr_s"] = st["arr_s"].astype(float)
    st["dep_s"] = st["dep_s"].astype(float)
    g = st.groupby("trip_id", sort=False)
    st["to_stop"] = g["stop_id"].shift(-1)
    st["t_s"] = (g["arr_s"].shift(-1) - st["dep_s"]).fillna(0.0)
    return st


def build_network(idx: ScheduleIndex, service_date: str, tz: str) -> Network:
    """Siec dnia `service_date` dla wszystkich linii aktywnych w efektywnym kalendarzu.

    Args:
        idx: rozklad z `index_schedule`.
        service_date: data obslugi YYYY-MM-DD.
        tz: strefa czasowa feedu.

    Returns:
        Siec dnia; zakres linii wybiera sie pozniej (`select_trips`).

    Raises:
        ValueError: w dniu nie ma aktywnych kursow.
    """
    ymd = service_date.replace("-", "")
    trips = _day_trips(idx, ymd)
    st = _day_stop_times(idx, trips)
    has = st["to_stop"].notna()
    pairs = st.loc[has, ["stop_id", "to_stop"]].set_axis(["from_stop", "to_stop"], axis=1)
    segments = pairs.drop_duplicates().sort_values(["from_stop", "to_stop"]).reset_index(drop=True)
    seg_ix = pd.MultiIndex.from_frame(segments)
    st["seg_id"] = -1
    st.loc[has, "seg_id"] = seg_ix.get_indexer(pd.MultiIndex.from_frame(pairs))
    counts = st.groupby("trip_id", sort=False).size()
    trips["n_rows"] = trips["trip_id"].map(counts).fillna(0).astype(int)
    trips["row_start"] = np.r_[0, np.cumsum(trips["n_rows"].to_numpy())[:-1]]
    segments.index.name = "seg_id"
    day_start = float(gtfs_day_start(pd.Series([ymd]), tz).iloc[0])
    cols = ["trip_id", "stop_sequence", "stop_id", "arr_s", "dep_s", "t_s", "seg_id"]
    return Network(service_date, day_start, trips.set_index("trip_id"), st[cols], segments)


def select_trips(net: Network, routes: list[str] | None) -> np.ndarray:
    """Pozycje (w `net.trips`) kursow linii z `routes`; None = wszystkie linie."""
    if routes is None:
        return np.arange(len(net.trips))
    return np.flatnonzero(net.trips["route_id"].astype(str).isin(routes).to_numpy())


def chain_order(net: Network, sel: np.ndarray) -> pd.DataFrame:
    """Kursy z wyboru z pozycja w lancuchu i poprzednikiem w wyborze.

    Returns:
        Ramka `trip` (pozycja w `net.trips`), `pos` (numer kursu w lancuchu w obrebie wyboru),
        `prev_sel` (pozycja poprzedniego kursu bloku, gdy jest w wyborze, inaczej -1).
    """
    sel = np.sort(sel)
    trips = net.trips
    at = pd.Series(np.arange(len(trips)), index=trips.index)
    prev = trips["prev_trip"].map(at).to_numpy(float)[sel]
    chosen = np.zeros(len(trips), dtype=bool)
    chosen[sel] = True
    ok = ~np.isnan(prev)
    prev_sel = np.full(len(sel), -1)
    prev_sel[ok] = np.where(chosen[prev[ok].astype(int)], prev[ok].astype(int), -1)
    out = pd.DataFrame({"trip": sel, "block": trips["block_key"].to_numpy()[sel]})
    out["pos"] = out.groupby("block", sort=False).cumcount()
    out["prev_sel"] = prev_sel
    return out[["trip", "pos", "prev_sel"]]
```

```python
"""Skladniki losowe przebiegu losowane z gory, wspolne dla obu silnikow."""

from dataclasses import dataclass

import numpy as np

from krakbus.sim.network import Network
from krakbus.sim.params import ModelParams


@dataclass(frozen=True)
class Noise:
    """Wszystkie liczby losowe jednego przebiegu.

    Attributes:
        e_d: efekt dnia [s na odcinek].
        delta: resztkowe spoznienie odjazdu z petli, po jednym na kurs (kolejnosc `net.trips`).
        eta: szum indywidualny, po jednym na wiersz `net.stop_times` (odcinek wychodzacy).
        u: zaklocenie wspolne [odcinek, okno].
    """

    e_d: float
    delta: np.ndarray
    eta: np.ndarray
    u: np.ndarray


def ar1(
    rng: np.random.Generator, n_series: int, n_steps: int, sigma: float, rho: float
) -> np.ndarray:
    """Niezalezne szeregi AR(1) o wariancji stacjonarnej `sigma**2` i korelacji `rho` miedzy krokami."""
    z = rng.standard_normal((n_series, n_steps))
    out = np.empty_like(z)
    out[:, 0] = sigma * z[:, 0]
    innov = sigma * np.sqrt(1.0 - rho**2)
    for k in range(1, n_steps):
        out[:, k] = rho * out[:, k - 1] + innov * z[:, k]
    return out


def draw_noise(net: Network, model: ModelParams, seed: int) -> Noise:
    """Losuje szum dla calej sieci dnia, wiec kurs dostaje ten sam szum niezaleznie od zakresu.

    Kolejnosc losowan (e_d, delta, eta, u) jest stala; zmiana parametru skali nie zmienia
    pozostalych skladnikow.
    """
    rng = np.random.default_rng(seed)
    e_d = float(rng.normal(0.0, model.day_effect_sd_s))
    delta = rng.normal(model.loop.delta_mean_s, model.loop.delta_sd_s, len(net.trips))
    eta = rng.standard_t(model.eta.df, len(net.stop_times)) * model.eta.scale_s
    n_win = int(net.stop_times["dep_s"].max() // model.u.window_s) + 1
    u = ar1(rng, len(net.segments), n_win, model.u.sigma_s, model.u.rho)
    return Noise(e_d=e_d, delta=delta, eta=eta, u=u)
```

```python
"""Rownania modelu wspolne dla obu silnikow: przyrost opoznienia na odcinku i regula petli."""

import numpy as np

from krakbus.sim.network import Network
from krakbus.sim.noise import Noise
from krakbus.sim.params import OFF_PEAK, ModelParams


def base_increments(net: Network, noise: Noise, model: ModelParams) -> np.ndarray:
    """Przyrost opoznienia na odcinku wychodzacym z kazdego wiersza `stop_times`, bez skladnika gamma.

    a_p * t_s / 60 + e_d + u[odcinek, okno] + eta; okno i pora dnia wg rozkladowego odjazdu
    z przystanku. Na ostatnim przystanku kursu 0.
    """
    st = net.stop_times
    dep = st["dep_s"].to_numpy(float)
    hour = (dep // 3600) % 24
    drift = np.full(len(st), model.drift_s_per_min[OFF_PEAK])
    for name, (h0, h1) in model.periods.items():
        drift[(hour >= h0) & (hour < h1)] = model.drift_s_per_min[name]
    seg = st["seg_id"].to_numpy()
    has = seg >= 0
    u = np.zeros(len(st))
    u[has] = noise.u[seg[has], (dep[has] // model.u.window_s).astype(int)]
    inc = drift * st["t_s"].to_numpy(float) / 60 + noise.e_d + u + noise.eta
    inc[~has] = 0.0
    return inc


def loop_departure(
    end_delay: np.ndarray | float,
    layover_s: np.ndarray | float,
    l_min_s: float,
    delta: np.ndarray | float,
) -> np.ndarray | float:
    """Opoznienie odjazdu nastepnego kursu: max(opoznienie na koncu + l_min - zapas, delta)."""
    return np.maximum(end_delay + l_min_s - layover_s, delta)
```

```python
"""Warunek brzegowy: opoznienie startowe kursu, ktorego poprzednik w bloku jest spoza zakresu."""

import numpy as np
import pandas as pd

from krakbus.sim.model import loop_departure
from krakbus.sim.network import Network
from krakbus.sim.noise import Noise
from krakbus.sim.params import ModelParams


def observed_end_delays(chains: pd.DataFrame) -> pd.Series:
    """Opoznienie przyjazdu na koniec kursu z lancuchow dnia (mediana, gdy kilka pojazdow)."""
    c = chains.dropna(subset=["arr_end_delay_s"])
    return c.groupby("trip_id")["arr_end_delay_s"].median().astype(float)


def boundary_starts(
    net: Network,
    sel: np.ndarray,
    noise: Noise,
    model: ModelParams,
    end_delays: pd.Series | None,
) -> dict[int, float]:
    """Opoznienie odjazdu kursow z wyboru, ktorych poprzednik w bloku jest spoza wyboru.

    Args:
        net: siec dnia.
        sel: pozycje kursow w zakresie.
        noise: szum przebiegu (`delta` dla kursow bez obserwacji).
        model: parametry (regula petli).
        end_delays: obserwowane opoznienie na koncu kursu wg `trip_id`; None = zawsze `delta`.

    Returns:
        Pozycja kursu -> Δ0 z reguly petli zastosowanej do obserwacji poprzednika albo `delta`.
    """
    trips = net.trips
    at = pd.Series(np.arange(len(trips)), index=trips.index)
    prev = trips["prev_trip"].map(at).to_numpy(float)
    chosen = np.zeros(len(trips), dtype=bool)
    chosen[sel] = True
    has_prev = ~np.isnan(prev)
    outside = np.zeros(len(trips), dtype=bool)
    outside[has_prev] = ~chosen[prev[has_prev].astype(int)]
    cand = np.flatnonzero(chosen & outside)
    delta = noise.delta[cand]
    if end_delays is None or end_delays.empty:
        return dict(zip(cand.tolist(), delta.tolist()))
    obs = trips["prev_trip"].iloc[cand].map(end_delays).to_numpy(float)
    lay = trips["layover_in_s"].to_numpy(float)[cand]
    start = np.where(
        np.isnan(obs), delta, loop_departure(obs, lay, model.loop.l_min_s, delta)
    )
    return dict(zip(cand.tolist(), start.tolist()))
```

```python
"""Silnik SimPy: jeden proces na lancuch kursow (blok) z zakresu."""

from collections.abc import Generator

import numpy as np
import simpy

from krakbus.sim.model import base_increments, loop_departure
from krakbus.sim.network import Network, chain_order
from krakbus.sim.noise import Noise
from krakbus.sim.params import ModelParams


def run_simpy(
    net: Network,
    sel: np.ndarray,
    noise: Noise,
    model: ModelParams,
    starts: dict[int, float],
) -> np.ndarray:
    """Doba w SimPy dla kursow z wyboru.

    Args:
        net: siec dnia.
        sel: pozycje kursow w zakresie.
        noise: szum przebiegu.
        model: parametry; przy `gamma` != 0 przyrost dostaje skladnik od bledu odstepu do
            poprzedniego autobusu tej samej linii i kierunku na przystanku.
        starts: Δ0 kursow z poprzednikiem spoza zakresu (`boundary_starts`).

    Returns:
        Opoznienie przyjazdu (= odjazdu) dla kazdego wiersza `net.stop_times`; NaN poza wyborem.
    """
    inc = base_increments(net, noise, model)
    st, trips = net.stop_times, net.trips
    arr_s = st["arr_s"].to_numpy(float)
    stop = st["stop_id"].to_numpy()
    row_start = trips["row_start"].to_numpy()
    n_rows = trips["n_rows"].to_numpy()
    line = list(zip(trips["route_id"].to_numpy(), trips["direction_id"].to_numpy()))
    layover = trips["layover_in_s"].to_numpy(float)
    l_min, gamma = model.loop.l_min_s, model.gamma
    delay = np.full(len(st), np.nan)
    last: dict[tuple, tuple[float, float]] = {}
    env = simpy.Environment()

    def vehicle(chain: list[tuple[int, int]]) -> Generator[simpy.Event, None, None]:
        end_delay = np.nan
        for p, prev_sel in chain:
            if prev_sel >= 0:
                d = float(loop_departure(end_delay, layover[p], l_min, noise.delta[p]))
            else:
                d = starts.get(p, float(noise.delta[p]))
            for j in range(row_start[p], row_start[p] + n_rows[p]):
                t = arr_s[j] + d
                yield env.timeout(max(0.0, t - env.now))
                delay[j] = d
                step = inc[j]
                if gamma:
                    key = (*line[p], stop[j])
                    if key in last:
                        t_prev, s_prev = last[key]
                        step += gamma * ((t - t_prev) - (arr_s[j] - s_prev))
                    last[key] = (t, arr_s[j])
                d = d + step
            end_delay = delay[row_start[p] + n_rows[p] - 1]

    order = chain_order(net, sel)
    block = net.trips["block_key"].to_numpy()[order["trip"].to_numpy()]
    for _, g in order.groupby(block, sort=False):
        env.process(vehicle(list(zip(g["trip"].tolist(), g["prev_sel"].tolist()))))
    env.run()
    return delay
```

```python
"""Komenda `simulate`: doba obslugi w DES dla zakresu linii, wynik w schemacie zdarzen obserwowanych."""

import argparse
import json
import logging
import subprocess
import time
from dataclasses import asdict
from pathlib import Path

import numpy as np
import pandas as pd

from krakbus.config import ROOT, load_config
from krakbus.ingest.gtfs_versions import load_schedule
from krakbus.logger import setup_logging
from krakbus.observed.events import index_schedule
from krakbus.observed.manifest import atomic_write_bytes, atomic_write_parquet, manifest_hash
from krakbus.observed.store import day_path
from krakbus.sim.boundary import boundary_starts, observed_end_delays
from krakbus.sim.engine import run_simpy
from krakbus.sim.network import Network, build_network, select_trips
from krakbus.sim.noise import draw_noise
from krakbus.sim.params import DEFAULT_SIM_CONFIG, SimConfig, load_sim_config

ENGINES = {"simpy": run_simpy}
SIM_COLUMNS = [
    "service_date", "vehicle_id", "trip_id", "route_id", "direction_id", "stop_sequence",
    "stop_id", "arr_s", "dep_s", "arr_ts", "dep_ts", "arr_delay_s", "dep_delay_s",
    "source", "run_id", "seed",
]
logger = logging.getLogger("simulate")


def scope_routes(sim_cfg: SimConfig, scope: str, routes: list[str] | None) -> list[str] | None:
    """Linie przebiegu: jawna lista albo zakres z configu (None = cala siec)."""
    if routes:
        return [str(r) for r in routes]
    if scope not in sim_cfg.scopes:
        raise ValueError(f"nieznany zakres {scope}; dostepne: {sorted(sim_cfg.scopes)}")
    return sim_cfg.scopes[scope]


def to_events(net: Network, sel: np.ndarray, delay: np.ndarray, run_id: str, seed: int) -> pd.DataFrame:
    """Zdarzenia symulacji w schemacie zblizonym do `data/processed/events/`."""
    st = net.stop_times
    rows = np.flatnonzero(~np.isnan(delay))
    ev = st.iloc[rows][["trip_id", "stop_sequence", "stop_id", "arr_s", "dep_s"]].copy()
    trips = net.trips.loc[ev["trip_id"]]
    d = delay[rows]
    ev["service_date"] = net.service_date
    ev["vehicle_id"] = trips["block_key"].to_numpy()
    ev["route_id"] = trips["route_id"].to_numpy()
    ev["direction_id"] = trips["direction_id"].to_numpy()
    ev["arr_ts"] = net.day_start + ev["arr_s"].to_numpy() + d
    ev["dep_ts"] = net.day_start + ev["dep_s"].to_numpy() + d
    ev["arr_delay_s"] = d
    ev["dep_delay_s"] = d
    ev["source"] = "sim"
    ev["run_id"] = run_id
    ev["seed"] = seed
    return ev[SIM_COLUMNS].reset_index(drop=True)


def simulate(
    net: Network,
    sim_cfg: SimConfig,
    routes: list[str] | None,
    label: str,
    seed: int,
    engine: str,
    end_delays: pd.Series | None,
) -> tuple[pd.DataFrame, dict]:
    """Jeden przebieg doby; zwraca zdarzenia i metadane (bez zapisu na dysk)."""
    t0 = time.perf_counter()
    sel = select_trips(net, routes)
    if len(sel) == 0:
        raise ValueError(f"brak kursow linii {routes} w dniu {net.service_date}")
    noise = draw_noise(net, sim_cfg.model, seed)
    obs = end_delays if sim_cfg.boundary == "observed" else None
    starts = boundary_starts(net, sel, noise, sim_cfg.model, obs)
    delay = ENGINES[engine](net, sel, noise, sim_cfg.model, starts)
    run_id = f"{net.service_date}_{'all' if routes is None else '-'.join(routes)}_{engine}_s{seed}"
    ev = to_events(net, sel, delay, run_id, seed)
    meta = {
        "run_id": run_id,
        "service_date": net.service_date,
        "routes": routes,
        "engine": engine,
        "seed": seed,
        "boundary": sim_cfg.boundary,
        "n_trips": int(len(sel)),
        "n_boundary": len(starts),
        "n_events": int(len(ev)),
        "elapsed_s": round(time.perf_counter() - t0, 3),
        "model": asdict(sim_cfg.model),
    }
    return ev, meta


def _git_commit() -> str:
    out = subprocess.run(
        ["git", "rev-parse", "--short", "HEAD"], cwd=ROOT, capture_output=True, text=True, check=False
    )
    return out.stdout.strip()


def _end_delays(processed: Path, day: str) -> pd.Series | None:
    path = day_path(processed, day, "chains")
    if not path.exists():
        logger.warning("brak %s; warunek brzegowy z delta", path)
        return None
    return observed_end_delays(pd.read_parquet(path, columns=["trip_id", "arr_end_delay_s"]))


def main() -> None:
    """Punkt wejscia `uv run simulate`."""
    parser = argparse.ArgumentParser(description=__doc__.splitlines()[0])
    parser.add_argument("--date", required=True, help="data obslugi YYYY-MM-DD")
    parser.add_argument("--scope", default="pilot", help="pilot | corridor | city")
    parser.add_argument("--routes", nargs="*", help="jawna lista linii (zamiast zakresu)")
    parser.add_argument("--seed", type=int, help="ziarno (domyslnie z configu)")
    parser.add_argument("--engine", default="simpy", choices=sorted(ENGINES))
    parser.add_argument("--config", type=Path, default=DEFAULT_SIM_CONFIG)
    args = parser.parse_args()
    setup_logging()
    cfg = load_config()
    sim_cfg = load_sim_config(args.config)
    routes = scope_routes(sim_cfg, args.scope, args.routes)
    seed = sim_cfg.seed if args.seed is None else args.seed
    sched = load_schedule(cfg.data.raw_dir / "static")
    net = build_network(index_schedule(sched), args.date, cfg.data.timezone)
    processed = cfg.data.processed_dir
    end_delays = _end_delays(processed, args.date) if sim_cfg.boundary == "observed" else None
    ev, meta = simulate(net, sim_cfg, routes, seed, args.engine, end_delays)
    meta |= {
        "scope": args.scope,
        "gtfs_versions": sched.versions,
        "manifest_hash": manifest_hash(processed / "manifest.json"),
        "git_commit": _git_commit(),
    }
    out = processed / "sim" / meta["run_id"]
    atomic_write_parquet(ev, out / "events.parquet")
    atomic_write_bytes(out / "run.json", json.dumps(meta, indent=1, ensure_ascii=False).encode())
    logger.info(
        "%s: kursow %d (brzegowych %d), zdarzen %d, %.1f s; mediana opoznienia %.0f s",
        meta["run_id"], meta["n_trips"], meta["n_boundary"], meta["n_events"],
        meta["elapsed_s"], float(np.median(ev["arr_delay_s"])),
    )
    logger.info("zapisano %s", out)
```
